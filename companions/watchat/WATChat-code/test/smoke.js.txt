'use strict';
const assert=require('assert'),fs=require('fs'),os=require('os'),path=require('path');
const {spawn}=require('child_process');const {WebSocket}=require('ws');
const port=19000+Math.floor(Math.random()*5000),origin='http://127.0.0.1:'+port,dataDir=fs.mkdtempSync(path.join(os.tmpdir(),'watchat-test-'));
const child=spawn(process.execPath,['backend/server.js'],{cwd:process.cwd(),env:Object.assign({},process.env,{PORT:String(port),ADMIN_PASSWORD:'admin-pass',DATA_DIR:dataDir}),stdio:['ignore','pipe','inherit']});
function ready(){return new Promise((ok,no)=>{const t=setTimeout(()=>no(new Error('start timeout')),5000);child.stdout.on('data',c=>{if(c.toString().includes('Watchat:')){clearTimeout(t);ok()}});child.once('exit',n=>no(new Error('server exited '+n)))})}
async function request(url,opt){const r=await fetch(origin+url,opt),x=await r.json();return{status:r.status,data:x}}
function auth(token,method,body){return{method:method||'GET',headers:Object.assign({Authorization:'Bearer '+token},body?{'Content-Type':'application/json'}:{}),body:body?JSON.stringify(body):undefined}}
function login(username,password){return request('/api/login',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({username,password})})}
function open(token){return new Promise((ok,no)=>{const ws=new WebSocket('ws://127.0.0.1:'+port+'/ws?token='+token);ws.inbox=[];ws.waiters=[];ws.on('message',raw=>{const x=JSON.parse(raw),i=ws.waiters.findIndex(w=>w.type===x.type);if(i<0)ws.inbox.push(x);else ws.waiters.splice(i,1)[0].ok(x)});ws.once('open',()=>ok(ws));ws.once('error',no)})}
function next(ws,type,timeout){return new Promise((ok,no)=>{const i=ws.inbox.findIndex(x=>x.type===type);if(i>=0)return ok(ws.inbox.splice(i,1)[0]);const waiter={type:type,ok:x=>{clearTimeout(t);ok(x)}},t=setTimeout(()=>{ws.waiters=ws.waiters.filter(w=>w!==waiter);no(new Error('timeout '+type))},timeout||2000);ws.waiters.push(waiter)})}
(async()=>{let sockets=[],uploadPath;try{await ready();let r=await login('admin','wrong');assert.equal(r.status,401);r=await login('admin','admin-pass');assert.equal(r.status,200);const admin=r.data.token,users={};for(const name of ['alice','bob','charlie']){r=await request('/api/admin/users',auth(admin,'POST',{username:name,name:name.toUpperCase(),password:'pass-'+name}));assert.equal(r.status,201);users[name]=r.data.user}r=await request('/api/admin/contact',auth(admin,'PATCH',{userId:users.alice.id,contactId:users.bob.id,enabled:true}));assert.equal(r.status,200);
const alice=(await login('alice','pass-alice')).data,bob=(await login('bob','pass-bob')).data,charlie=(await login('charlie','pass-charlie')).data;assert.deepEqual(alice.contacts.map(x=>x.username),['bob']);assert.deepEqual(bob.contacts.map(x=>x.username),['alice']);assert.equal(charlie.contacts.length,0);
const aw=await open(alice.token),bw=await open(bob.token),cw=await open(charlie.token);sockets=[aw,bw,cw];await Promise.all(sockets.map(ws=>next(ws,'ready')));aw.send(JSON.stringify({action:'history',peerId:users.bob.id}));assert.equal((await next(aw,'history')).messages.length,0);
let incoming=next(bw,'message');aw.send(JSON.stringify({action:'send',to:users.bob.id,kind:'text',text:'你好 Bob'}));let msg=(await incoming).message;assert.equal(msg.text,'你好 Bob');assert.equal(msg.from,users.alice.id);
aw.send(JSON.stringify({action:'send',to:users.charlie.id,kind:'text',text:'不应送达'}));let blocked=false;try{await next(cw,'message',350)}catch(_){blocked=true}assert.equal(blocked,true);
r=await fetch(origin+'/api/upload?type=audio',{method:'POST',headers:{Authorization:'Bearer '+alice.token,'Content-Type':'audio/webm'},body:Buffer.from('voice')});assert.equal(r.status,201);const media=await r.json();uploadPath=path.join(process.cwd(),media.url);incoming=next(bw,'message');aw.send(JSON.stringify({action:'send',to:users.bob.id,kind:'audio',url:media.url,duration:3.4}));msg=(await incoming).message;assert.equal(msg.kind,'audio');assert.equal(msg.duration,3.4);
r=await request('/api/session',auth(alice.token));assert.equal(r.data.user.username,'alice');assert.equal(fs.existsSync(path.join(dataDir,'store.json')),true);console.log('Smoke passed: users, symmetric contacts, ACL, persistence, text/audio WebSocket');
}finally{sockets.forEach(ws=>ws.close());child.kill('SIGTERM');if(uploadPath)try{fs.unlinkSync(uploadPath)}catch(_){}fs.rmSync(dataDir,{recursive:true,force:true})}})().catch(e=>{console.error(e);process.exitCode=1});
